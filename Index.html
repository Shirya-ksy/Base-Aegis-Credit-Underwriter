// SPDX-License-Identifier: MIT pragma solidity 0.8.36; contract AegisCreditUnderwriter { struct BorrowerProfile { uint256 creditScore; // 300 - 850 uint256 maxCreditLine; uint256 activeUnderwrittenETH; bool isApproved; } address payable public immutable underwriterMaster; string public protocolDescriptor; uint256 public baseLeverageRatio; // например, 200 = 2.0x uint256 public totalReserveCapital; uint256 public lockedCapital; uint256 public defaultFundBalance; mapping(address => BorrowerProfile) public borrowers; event ReserveFunded(address indexed guarantor, uint256 amount); event ScoreEvaluated(address indexed borrower, uint256 score, uint256 maxCredit); event CollateralUnderwritten(address indexed borrower, uint256 lockedAmount); event CollateralReleased(address indexed borrower, uint256 releasedAmount); event CollateralSlashed(address indexed borrower, uint256 slashedAmount); event MetadataUpdated(string newDescriptor, uint256 newLeverage); event ReserveWithdrawn(address indexed recipient, uint256 amount); modifier onlyMaster() { require(msg.sender == underwriterMaster, "AegisCredit: Caller is not underwriter master"); _; } constructor() { underwriterMaster = payable(msg.sender); protocolDescriptor = "Aegis Base On-Chain Credit Registry v1"; baseLeverageRatio = 200; // 2.0x } // 1. Пополнение резервного капитала поручительства function depositGuarantorReserve() external payable onlyMaster { require(msg.value > 0, "AegisCredit: Zero deposit forbidden"); totalReserveCapital += msg.value; emit ReserveFunded(msg.sender, msg.value); } // 2. Оценка и присвоение кредитного скоринга function evaluateCreditScore(address borrower, uint256 score, uint256 creditLine) external onlyMaster { require(borrower != address(0), "AegisCredit: Zero borrower address"); require(score >= 300 && score <= 850, "AegisCredit: Score out of range (300-850)"); borrowers[borrower] = BorrowerProfile({ creditScore: score, maxCreditLine: creditLine, activeUnderwrittenETH: borrowers[borrower].activeUnderwrittenETH, isApproved: true }); emit ScoreEvaluated(borrower, score, creditLine); } // 3. Заморозка гарантийного фонда под сделку заёмщика function lockUnderwritingCollateral(address borrower, uint256 amount) external onlyMaster { BorrowerProfile storage bp = borrowers[borrower]; require(bp.isApproved, "AegisCredit: Borrower not evaluated"); require(bp.activeUnderwrittenETH + amount <= bp.maxCreditLine, "AegisCredit: Exceeds approved line"); require(lockedCapital + amount <= totalReserveCapital, "AegisCredit: Insufficient unreserved capital"); bp.activeUnderwrittenETH += amount; lockedCapital += amount; emit CollateralUnderwritten(borrower, amount); } // 4. Штатное освобождение гарантийного залога function releaseUnderwritingCollateral(address borrower, uint256 amount) external onlyMaster { BorrowerProfile storage bp = borrowers[borrower]; require(bp.activeUnderwrittenETH >= amount, "AegisCredit: Amount exceeds active underwritten"); bp.activeUnderwrittenETH -= amount; lockedCapital -= amount; emit CollateralReleased(borrower, amount); } // 5. Конфискация залога в фонд покрытия дефолтов function slashDefaultedCollateral(address borrower, uint256 amount) external onlyMaster { BorrowerProfile storage bp = borrowers[borrower]; require(bp.activeUnderwrittenETH >= amount, "AegisCredit: Slashed amount exceeds underwritten"); bp.activeUnderwrittenETH -= amount; lockedCapital -= amount; totalReserveCapital -= amount; defaultFundBalance += amount; emit CollateralSlashed(borrower, amount); } // 6. Обновление метаданных протокола function updateProtocolMetadata(string calldata newDescriptor, uint256 newLeverage) external onlyMaster { require(bytes(newDescriptor).length > 0, "AegisCredit: Empty descriptor"); require(newLeverage >= 100 && newLeverage <= 500, "AegisCredit: Leverage 1x - 5x"); protocolDescriptor = newDescriptor; baseLeverageRatio = newLeverage; emit MetadataUpdated(newDescriptor, newLeverage); } // 7. Вывод незадействованного гарантийного резерва и средств дефолтного фонда function withdrawGuarantorFunds(uint256 amount) external onlyMaster { uint256 available = (totalReserveCapital - lockedCapital) + defaultFundBalance; require(amount <= available && amount > 0, "AegisCredit: Invalid withdrawal amount"); if (amount <= defaultFundBalance) { defaultFundBalance -= amount; } else { uint256 fromReserve = amount - defaultFundBalance; defaultFundBalance = 0; totalReserveCapital -= fromReserve; } (bool sent, ) = underwriterMaster.call{value: amount}(""); require(sent, "AegisCredit: Withdrawal failed"); emit ReserveWithdrawn(underwriterMaster, amount); } // Просмотр профиля заёмщика function getBorrowerStatus(address borrower) external view returns ( uint256 creditScore, uint256 maxCreditLine, uint256 activeUnderwrittenETH, bool isApproved ) { BorrowerProfile memory bp = borrowers[borrower]; return (bp.creditScore, bp.maxCreditLine, bp.activeUnderwrittenETH, bp.isApproved); } }
